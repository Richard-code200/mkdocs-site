sudo groupadd police
sudo groupadd bandit

sudo useradd -m -g police jack
sudo useradd -m -g police jerry
sudo useradd -m -g bandit xh
sudo useradd -m -g bandit xq

su -jack
touch secret.txt
chmod 640 secret.txt
ls -l secret.txt

chmod 664 secret.txt
ls -l secret.txt

sudo usermod -g police xh
su - xh
cat /home/jack/secret.txt
echo "xh was here" >> /home/jack/secret.txt
id xh
